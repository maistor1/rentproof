# RentProof Backend Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the RentProof backend (photo upload and serving, rental return with mocked telematics, finding review, invoice) on storage that survives parallel uploads.

**Architecture:** Route handlers stay thin: parse and validate, call a function in `src/lib`, map `HttpError` to `{ error }` JSON. All state lives in JSON files under `data/`. Every read-modify-write goes through one process-wide queue, and every write is atomic (temp file + rename). `data/rentals.json` becomes a read-only seed; runtime changes go to the gitignored `data/rentals.local.json`.

**Tech Stack:** Next.js 16.4 route handlers, Node 24, TypeScript, sharp 0.35, Vitest 5.

**Spec:** `/Users/bori/RentProof — Hackathon Build Plan.md` (RP-5, RP-7, RP-11, decision rules, price table) and `README.md` (API surface, contracts). Use the `nextjs16-route-handlers` skill for every route file.

## Global Constraints

- `src/lib/types.ts` does not change. It is the team contract; every response shape comes from it.
- No new runtime dependencies. sharp, Node built-ins and Next only.
- Errors are `{ error: string }` with status 400 bad input, 404 unknown id, 409 wrong state, 413 too large, 415 unsupported image.
- Photos: long edge at most 1,500 px, JPEG, auto-rotated from EXIF, EXIF stripped.
- Prices come only from `data/price-table.json`, never from the model.
- Billable findings are exactly `auto_flagged` and `confirmed`.
- Route params are promises typed with `RouteContext<"...">`; run `npx next typegen` after adding a route folder.
- Every file under `src/lib` that touches the filesystem starts with `import "server-only";`.
- Each task ends with `npm test`, `npx tsc --noEmit` and `npm run lint` passing before the commit.

## Review Focus

1. **Six photos uploaded at once to one rental:** every slot keeps its id. Pinned by the parallel tests in Task 1 and Task 2.
2. **An iPhone HEIC or other non-JPEG file:** the worker gets a 415 with a readable message, not a 500. Task 3 pins it with non-image bytes. True HEIC needs a manual check with a real iPhone photo (Task 7, Step 6).
3. **Ids containing `../` in URLs:** 404, and nothing outside `data/` is read. Pinned in Task 1 (storage) and Task 3 (route).
4. **"Finish return" tapped twice:** the rental is returned once, and the hours aren't added twice. Pinned in Task 4.
5. **Staff confirms a finding after the invoice was opened:** the next invoice fetch includes it, because invoices are computed on demand and never cached. Pinned in Task 6.

```ts
import "server-only";
import { randomUUID } from "node:crypto";
import { mkdir, readFile, rename, writeFile } from "node:fs/promises";
import path from "node:path";
import type { DamageReport, Photo, Rental } from "@/lib/types";

// All state lives in JSON files under data/ (RENTPROOF_DATA_DIR in tests).
// rentals.json is the tracked seed. The app writes rentals.local.json, which
// is gitignored, so demo runs never dirty the git tree.
export const dataDir = () => process.env.RENTPROOF_DATA_DIR ?? path.join(process.cwd(), "data");
export const photoDir = () => path.join(dataDir(), "photos");
const inspectionDir = () => path.join(dataDir(), "inspections");
const seedFile = () => path.join(dataDir(), "rentals.json");
const rentalsFile = () => path.join(dataDir(), "rentals.local.json");

// Ids end up in file names, so only accept the characters we generate.
const SAFE_ID = /^[A-Za-z0-9_-]{1,64}$/;
export const isSafeId = (id: string) => SAFE_ID.test(id);
export const newId = (prefix: string) => `${prefix}-${randomUUID().slice(0, 8)}`;

// Runs read-modify-write steps one at a time. The queue lives on globalThis
// because Next can load this module more than once (one copy per route bundle
// in dev), and separate queues would not see each other.
const shared = globalThis as { __rentproofQueue?: Promise<unknown> };
export function serialized<T>(fn: () => Promise<T>): Promise<T> {
  const run = (shared.__rentproofQueue ?? Promise.resolve()).then(fn);
  shared.__rentproofQueue = run.catch(() => undefined);
  return run;
}

// Readers never see a half-written file: write a temp file, then rename.
async function writeJson(file: string, value: unknown): Promise<void> {
  await mkdir(path.dirname(file), { recursive: true });
  const tmp = `${file}.${randomUUID()}.tmp`;
  await writeFile(tmp, JSON.stringify(value, null, 2) + "\n");
  await rename(tmp, file);
}

async function readJson<T>(file: string): Promise<T | undefined> {
  try {
    return JSON.parse(await readFile(file, "utf8"));
  } catch (e) {
    if ((e as NodeJS.ErrnoException).code === "ENOENT") return undefined;
    throw e;
  }
}

export async function listRentals(): Promise<Rental[]> {
  return (await readJson<Rental[]>(rentalsFile())) ?? (await readJson<Rental[]>(seedFile())) ?? [];
}

export async function getRental(id: string): Promise<Rental | undefined> {
  return (await listRentals()).find((r) => r.id === id);
}

// Applies `change` to the stored rental and saves it. Returns undefined if the
// rental doesn't exist. If `change` throws, nothing is written.
export function updateRental(
  id: string,
  change: (rental: Rental) => void,
): Promise<Rental | undefined> {
  return serialized(async () => {
    const rentals = await listRentals();
    const rental = rentals.find((r) => r.id === id);
    if (!rental) return undefined;
    change(rental);
    await writeJson(rentalsFile(), rentals);
    return rental;
  });
}

export function saveRental(rental: Rental): Promise<void> {
  return serialized(async () => {
    const rentals = await listRentals();
    const i = rentals.findIndex((r) => r.id === rental.id);
    if (i === -1) rentals.push(rental);
    else rentals[i] = rental;
    await writeJson(rentalsFile(), rentals);
  });
}

const reportFile = (id: string) => path.join(inspectionDir(), `${id}.json`);

export async function getReport(id: string): Promise<DamageReport | undefined> {
  return isSafeId(id) ? readJson<DamageReport>(reportFile(id)) : undefined;
}

export function saveReport(report: DamageReport): Promise<void> {
  return serialized(async () => {
    if (!isSafeId(report.id)) throw new Error(`unsafe report id: ${report.id}`);
    await writeJson(reportFile(report.id), report);
  });
}

export function updateReport(
  id: string,
  change: (report: DamageReport) => void,
): Promise<DamageReport | undefined> {
  return serialized(async () => {
    const report = await getReport(id);
    if (!report) return undefined;
    change(report);
    await writeJson(reportFile(id), report);
    return report;
  });
}

// Each photo is two write-once files, <id>.jpg and <id>.json, so uploads
// don't contend for a shared index.
export async function writePhoto(photo: Photo, jpeg: Buffer): Promise<void> {
  await mkdir(photoDir(), { recursive: true });
  await writeFile(path.join(photoDir(), photo.path), jpeg);
  await writeJson(path.join(photoDir(), `${photo.id}.json`), photo);
}

export async function getPhoto(id: string): Promise<Photo | undefined> {
  return isSafeId(id) ? readJson<Photo>(path.join(photoDir(), `${id}.json`)) : undefined;
}

export async function readPhoto(id: string): Promise<Buffer | undefined> {
  const photo = await getPhoto(id);
  return photo && readFile(path.join(photoDir(), photo.path));
}
```

```ts
// Errors the API reports to the caller as { error } with a status code.
// Anything else is a bug and surfaces as a 500.
export class HttpError extends Error {
  status: number;
  constructor(status: number, message: string) {
    super(message);
    this.status = status;
  }
}

export const jsonError = (status: number, message: string) =>
  Response.json({ error: message }, { status });

export function errorResponse(e: unknown): Response {
  if (e instanceof HttpError) return jsonError(e.status, e.message);
  throw e;
}
```

```ts
import "server-only";
import sharp from "sharp";
import { HttpError } from "@/lib/http";
import { getRental, newId, updateRental, writePhoto } from "@/lib/storage";
import type { Phase, Photo, Slot } from "@/lib/types";

export const MAX_EDGE = 1500;
export const MAX_UPLOAD_BYTES = 25 * 1024 * 1024;

// Turns the photo upright from its EXIF orientation (phones store rotation
// there), fits the long edge to 1,500 px and re-encodes as JPEG. sharp drops
// EXIF on output, which also removes the phone's GPS position. Anything sharp
// can't decode, including iPhone HEIC with the prebuilt binary, is a 415.
export async function normalizePhoto(input: Buffer) {
  try {
    const { data, info } = await sharp(input)
      .rotate()
      .resize({ width: MAX_EDGE, height: MAX_EDGE, fit: "inside", withoutEnlargement: true })
      .jpeg({ quality: 85 })
      .toBuffer({ resolveWithObject: true });
    return { jpeg: data, width: info.width, height: info.height };
  } catch {
    throw new HttpError(415, "unsupported image; send a JPEG or PNG photo");
  }
}

// RP-5: stores the photo and points the rental's slot at it. A retake gets a
// new id; the old file stays on disk for the log.
export async function savePhoto(input: {
  rentalId: string;
  slot: Slot;
  phase: Phase;
  file: Buffer;
}): Promise<Photo> {
  const { rentalId, slot, phase } = input;
  if (!(await getRental(rentalId))) throw new HttpError(404, "rental not found");

  const { jpeg, width, height } = await normalizePhoto(input.file);
  const id = newId("P");
  const photo: Photo = {
    id,
    rentalId,
    slot,
    phase,
    path: `${id}.jpg`,
    width,
    height,
    takenAt: new Date().toISOString(),
  };
  await writePhoto(photo, jpeg);
```

```ts
import { MAX_UPLOAD_BYTES, savePhoto } from "@/lib/photos";
import { SLOTS, type Phase, type Slot } from "@/lib/types";

const PHASES: readonly Phase[] = ["checkout", "return"];

// RP-5: multipart form with rentalId, slot, phase, file. Returns the Photo.
export async function POST(req: Request) {
  const form = await req.formData().catch(() => null);
  if (!form) return jsonError(400, "expected multipart form data");

  const rentalId = form.get("rentalId");
  const slot = form.get("slot");
  const phase = form.get("phase");
  const file = form.get("file");
  if (typeof rentalId !== "string" || !rentalId) return jsonError(400, "rentalId required");
  if (typeof slot !== "string" || !SLOTS.includes(slot as Slot)) {
    return jsonError(400, `slot must be one of: ${SLOTS.join(", ")}`);
  }
  if (typeof phase !== "string" || !PHASES.includes(phase as Phase)) {
    return jsonError(400, "phase must be checkout or return");
  }
  if (!(file instanceof File) || file.size === 0) return jsonError(400, "file required");
  if (file.size > MAX_UPLOAD_BYTES) return jsonError(413, "file too large (max 25 MB)");

  try {

      rentalId,
      slot: slot as Slot,
      phase: phase as Phase,
      file: Buffer.from(await file.arrayBuffer()),
    });
    return Response.json(photo);
  } catch (e) {
    return errorResponse(e);
  }
```

```ts
import { HttpError } from "@/lib/http";
import { updateRental } from "@/lib/storage";
import { mockSnapshot } from "@/lib/telematics";
import type { Rental } from "@/lib/types";

// Hours the mocked telematics feed reports at return unless the caller says
// otherwise. A real adapter would read them from the OEM's ISO 15143-3 API.
export const DEMO_HOURS_USED = 37.5;

function hours(value: unknown, field: string): number {
  if (typeof value !== "number" || !Number.isFinite(value) || value < 0) {
    throw new HttpError(400, `${field} must be a non-negative number`);
  }
  return value;
}

// RP-7: closes the rental with the hours from the (mocked) telematics feed.
// Calling it again returns the rental unchanged, so a double tap is harmless.
export async function recordReturn(
  rentalId: string,
  hoursUsed: unknown = DEMO_HOURS_USED,
): Promise<Rental> {
  const used = hours(hoursUsed, "hoursUsed");
  const rental = await updateRental(rentalId, (r) => {
    if (r.status !== "out") return;
    const snapshot = mockSnapshot(r, used);
    r.telematics = snapshot;
    r.hourMeterAtReturn = snapshot.CumulativeOperatingHours.Hour;
    r.returnAt = snapshot.CumulativeOperatingHours.datetime;
    r.status = "returned";
  });
  if (!rental) throw new HttpError(404, "rental not found");
  return rental;
}
```

```ts
import "server-only";
import { HttpError } from "@/lib/http";
import { updateReport } from "@/lib/storage";
import type { DamageReport, FindingStatus } from "@/lib/types";

export type ReviewDecision = "confirmed" | "rejected";

const DECISIONS: readonly string[] = ["confirmed", "rejected"];
// Staff decide only the 0.50-0.79 band. auto_flagged and hidden are set by the
// confidence rules and stay that way.
const REVIEWABLE: readonly FindingStatus[] = ["needs_review", "confirmed", "rejected"];

export async function reviewFinding(
  reportId: string,
  findingId: string,
  decision: unknown,
): Promise<DamageReport> {
  if (typeof decision !== "string" || !DECISIONS.includes(decision)) {
    throw new HttpError(400, "status must be confirmed or rejected");
  }
  const report = await updateReport(reportId, (r) => {
    const finding = r.findings.find((f) => f.id === findingId);
    if (!finding) throw new HttpError(404, "finding not found");
    if (!REVIEWABLE.includes(finding.status)) {
      throw new HttpError(409, `finding is ${finding.status}; only review-band findings can be changed`);
    }
    finding.status = decision as ReviewDecision;
  });
  if (!report) throw new HttpError(404, "report not found");
  return report;
}
```

```ts
import priceTable from "../../data/price-table.json";
import { HttpError } from "@/lib/http";
import type {
  DamageReport,
  DamageType,
  FindingStatus,
  Invoice,
  InvoiceLine,
  Rental,
  ReviewedFinding,
  Severity,
} from "@/lib/types";

export const PRICES = priceTable;

const BILLABLE: readonly FindingStatus[] = ["auto_flagged", "confirmed"];
const PRICE_BY_KEY: Record<string, number> = PRICES.prices;

const sentence = (s: string) => s.charAt(0).toUpperCase() + s.slice(1).replaceAll("_", " ");
const lineDescription = (f: ReviewedFinding) =>
  `${sentence(f.type)} (${f.severity}), ${f.slot.replaceAll("_", " ")}: ${f.location}`;

// RP-11: one line per auto_flagged or confirmed finding, priced from the
// sample table. The model never sets a price. Computed on every request, so
// staff reviews show up immediately.
export function buildInvoice(rental: Rental, report: DamageReport): Invoice {
  const lines: InvoiceLine[] = report.findings
    .filter((f) => BILLABLE.includes(f.status))
    .map((f) => {
      const priceKey: `${DamageType}/${Severity}` = `${f.type}/${f.severity}`;
      const amount = PRICE_BY_KEY[priceKey];
      if (amount === undefined) throw new Error(`no price for ${priceKey} in price-table.json`);
      const beforePhotoId = rental.photos.checkout[f.slot];
      const afterPhotoId = rental.photos.return[f.slot];
      if (!beforePhotoId || !afterPhotoId) {
        throw new HttpError(409, `finding ${f.id} has no ${f.slot} photo pair to show as proof`);
      }
      return { findingId: f.id, description: lineDescription(f), priceKey, amount, beforePhotoId, afterPhotoId };
    });
  const total = Math.round(lines.reduce((sum, l) => sum + l.amount, 0) * 100) / 100;
  return { reportId: report.id, rentalId: rental.id, currency: PRICES.currency, lines, total };
}
```

```ts
import { errorResponse, jsonError } from "@/lib/http";
import { buildInvoice } from "@/lib/pricing";

// RP-11: the Invoice for a saved DamageReport.
export async function GET(_req: Request, ctx: RouteContext<"/api/invoice/[inspectionId]">) {
  const { inspectionId } = await ctx.params;
  const report = await getReport(inspectionId);
  if (!report) return jsonError(404, "report not found");
  const rental = await getRental(report.rentalId);
  if (!rental) return jsonError(404, "rental not found");
  try {

  } catch (e) {
    return errorResponse(e);
  }
}
```
